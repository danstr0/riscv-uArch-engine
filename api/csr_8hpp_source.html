128 void write(u32 addr, u32 value) noexcept
129 {
130 switch (addr)
131 {
132 case CSRAddr::MSTATUS:
133 // Only MIE and MPIE are writable
134 mstatus_ = value & (MStatus::MIE | MStatus::MPIE);
135 break;
136 case CSRAddr::MIE:
137 mie_ = value & (MInterrupt::MSIE | MInterrupt::MTIE | MInterrupt::MEIE);
138 break;
139 case CSRAddr::MTVEC:
140 // WARL: MODE values >= 2 are reversed; treat them as DIRECT
141 if ((value & MTvecMode::MASK) > MTvecMode::VECTORED)
142 value &= ~MTvecMode::MASK;
143 mtvec_ = value;
144 break;
145 case CSRAddr::MSCRATCH:
146 mscratch_ = value;
147 break;
148 case CSRAddr::MEPC:
149 mepc_ = value & ~1u; // Bit 0 always zero (IALIGN=32)
150 break;
151 case CSRAddr::MCAUSE:
152 mcause_ = value;
153 break;
154 case CSRAddr::MIP:
155 // On an M-mode-only hart, MSIP/MTIP/MEIP are read-only in
156 // mip: they mirror the CLINT msip register, the CLINT
157 // timer comparator, and the PLIC, respectively. Software
158 // clears them at the source (write msip, write mtimecmp,
159 // claim at the PLIC)
160 break;
161 case CSRAddr::MHARTID:
162 // Read-only
163 break;
164 default:
165 break;
166 }
167 }

208 {
209 if (!(mstatus_ & MStatus::MIE)) return 0;
210
211 u32 pending = mip_ & mie_;
212 if (pending & MInterrupt::MEIP) return MCause::M_EXTERNAL;
213 if (pending & MInterrupt::MSIP) return MCause::M_SOFTWARE;
214 if (pending & MInterrupt::MTIP) return MCause::M_TIMER;

225 [[nodiscard]] addr_t trap_target(u32 cause) const noexcept
226 {
227 const addr_t base = mtvec_ & ~MTvecMode::MASK;
228 if ((mtvec_ & MTvecMode::MASK) == MTvecMode::VECTORED
229 && (cause & MCause::INTERRUPT_BIT))
230 return base + 4 * (cause & ~MCause::INTERRUPT_BIT);
231 return base;
