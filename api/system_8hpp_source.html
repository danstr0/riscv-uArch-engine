65public:
66 explicit SingleCoreSystem(SystemConfig config = {})
67 : config_(config)
68 {
69 validate();
70
71 config_.platform.num_harts = 1;
72 platform_ = std::make_unique<Platform>(config_.platform);
73
74 std::shared_ptr<Memory> shared = platform_->ram_ptr();
75 if (config_.l3.size_bytes) { l3_ = std::make_shared<Cache>(config_.l3, shared); shared = l3_; }
76 if (config_.l2.size_bytes) { l2_ = std::make_shared<Cache>(config_.l2, shared); shared = l2_; }
77
78 std::shared_ptr<Memory> data_side = shared;
79 if (config_.l1d.size_bytes)
80 {
81 config_.l1d.type = config_.l1i.size_bytes ? CacheType::DATA
82 : CacheType::UNIFIED;
83 l1d_ = std::make_shared<Cache>(config_.l1d, shared);
84 data_side = l1d_;
85 }
86
87 std::shared_ptr<Memory> fetch_side = data_side;
88 if (config_.l1i.size_bytes)
89 {
90 config_.l1i.type = CacheType::INSTRUCTION;
91 l1i_ = std::make_shared<Cache>(config_.l1i, shared);
92 fetch_side = l1i_;
93
94 // A refill must see code just written through the D-cache
95 l1i_->set_on_read_miss([d = l1d_.get()](addr_t line) -> u32
96 {
97 d->writeback(line);
98 return 0;
99 });
100 }
101
102 dma_ = make_dma_port(shared);
103
104 bus_ = platform_->make_hart_bus(data_side);
105 fetch_bus_ = platform_->make_fetch_bus(fetch_side);
106
107 if constexpr (std::is_same_v<Core, PipelinedCPU>)
108 core_ = std::make_unique<Core>(bus_, config_.pipeline, fetch_bus_);
109 else
110 core_ = std::make_unique<Core>(bus_, fetch_bus_);
111
112 if (l1i_)
113 core_->set_on_fence_i([i = l1i_.get()] { i->invalidate_all(); });
114
115 platform_->connect_hart(0, core_->csrs());

145 template <typename Pred>
146 u64 run_until(Pred&& pred, u64 max_steps = 10'000'000)
147 {
148 u64 n = 0;

198 void validate() const
199 {
200 if (config_.l1d.size_bytes == 0 && config_.l1i.size_bytes)
201 throw std::invalid_argument("SingleCoreSystem: L1I requires an L1D cache");
202 if (config_.l1d.size_bytes == 0
203 && (config_.l2.size_bytes || config_.l3.size_bytes))
204 throw std::invalid_argument("SingleCoreSystem: L2/L3 require an L1 cache");
205 if (config_.l2.size_bytes == 0 && config_.l3.size_bytes)
206 throw std::invalid_argument("SingleCoreSystem: L3 requires an L2 cache");
207 if (config_.l1i.size_bytes && config_.l1i.line_size != config_.l1d.line_size)
208 throw std::invalid_argument("SingleCoreSystem: L1I and L1D line sizes must match");
209 }

211 [[nodiscard]] std::shared_ptr<Memory>
212 make_dma_port(std::shared_ptr<Memory> shared)
213 {
214 if (!l1d_) return shared;
215
216 Cache* d = l1d_.get();
217 return std::make_shared<DmaPort>(
218 std::move(shared), config_.l1d.line_size,
219 [d](addr_t line) { d->writeback(line); },
220 [d](addr_t line) { d->snoop_invalidate(line); });
221 }
